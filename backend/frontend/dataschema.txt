-- Drop existing tables with cascade to ensure a clean sync
DROP TABLE IF EXISTS notifications CASCADE;
DROP TABLE IF EXISTS settlements CASCADE;
DROP TABLE IF EXISTS handovers CASCADE;
DROP TABLE IF EXISTS maintenance_tickets CASCADE;
DROP TABLE IF EXISTS inspections CASCADE;
DROP TABLE IF EXISTS documents CASCADE;
DROP TABLE IF EXISTS transactions CASCADE;
DROP TABLE IF EXISTS meter_readings CASCADE;
DROP TABLE IF EXISTS utility_charges CASCADE;
DROP TABLE IF EXISTS invoices CASCADE;
DROP TABLE IF EXISTS leases CASCADE;
DROP TABLE IF EXISTS owners CASCADE;
DROP TABLE IF EXISTS units CASCADE;
DROP TABLE IF EXISTS buildings CASCADE;
DROP TABLE IF EXISTS properties CASCADE;

-- 1. PROPERTIES TABLE
CREATE TABLE properties (
    id SERIAL PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    property_name VARCHAR(255),
    type VARCHAR(100) DEFAULT 'Residential Multi-Family',
    location TEXT,
    units_count INT DEFAULT 20,
    occupancy VARCHAR(20) DEFAULT '5.0%',
    status VARCHAR(50) DEFAULT 'Active'
);

-- 2. BUILDINGS TABLE
CREATE TABLE buildings (
    id SERIAL PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(100) NOT NULL,
    property_id INT REFERENCES properties(id) ON DELETE CASCADE,
    property_name VARCHAR(255),
    floors INT DEFAULT 5,
    total_units INT DEFAULT 20,
    occupied_units INT DEFAULT 1,
    status VARCHAR(50) DEFAULT 'ACTIVE'
);

-- 3. UNITS TABLE (20 Units in Tower A)
CREATE TABLE units (
    id SERIAL PRIMARY KEY,
    unit_number VARCHAR(50) NOT NULL UNIQUE,
    building_id INT REFERENCES buildings(id) ON DELETE SET NULL,
    property_name VARCHAR(255) DEFAULT 'Sunrise Residences • Tower A',
    floor INT,
    type VARCHAR(50) DEFAULT '1BR Executive',
    status VARCHAR(50) DEFAULT 'VACANT',
    tenant_name VARCHAR(255),
    monthly_rent NUMERIC(12, 2) DEFAULT 15000.00
);

-- 5. LEASES TABLE (Maria Santos • LSE-2026-0101)

    tenant_name VARCHAR(255) NOT NULL,

    end_date DATE DEFAULT '2026-12-31',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP

-- 6. INVOICES TABLE
CREATE TABLE invoices (
    id VARCHAR(50) PRIMARY KEY,
    invoice_id VARCHAR(50),
    lease_id VARCHAR(50) REFERENCES leases(lease_id) ON DELETE SET NULL,
    tenant_name VARCHAR(255),
    tenant_email VARCHAR(255),
    type VARCHAR(255) NOT NULL,
    sub TEXT,
    property_location TEXT,
    category_type VARCHAR(100),
    due_date DATE,
    amount NUMERIC(12, 2) NOT NULL,
    status VARCHAR(50) DEFAULT 'Unpaid',
    channel VARCHAR(100) DEFAULT 'Pending Payment',
    ref_no VARCHAR(100),
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 7. UTILITY CHARGES TABLE
CREATE TABLE utility_charges (
    id VARCHAR(50) PRIMARY KEY,
    charge_id VARCHAR(50),
    tenant_name VARCHAR(255),
    tenant_email VARCHAR(255),
    unit_location TEXT,
    type VARCHAR(100),
    breakdown TEXT,
    period VARCHAR(100),
    due_date DATE,
    amount NUMERIC(12, 2) NOT NULL,
    status VARCHAR(50) DEFAULT 'Pending',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 8. METER READINGS TABLE
CREATE TABLE meter_readings (
    id VARCHAR(50) PRIMARY KEY,
    reading_id VARCHAR(50),
    tenant_name VARCHAR(255),
    tenant_email VARCHAR(255),
    unit_location TEXT,
    utility VARCHAR(100),
    serial VARCHAR(100),
    prev_dial NUMERIC(12, 2),
    curr_dial NUMERIC(12, 2),
    consumption NUMERIC(12, 2),
    rate_per_unit NUMERIC(10, 2),
    total_billed_amount NUMERIC(12, 2),
    unit_symbol VARCHAR(20),
    period VARCHAR(100),
    reading_date DATE,
    status VARCHAR(100) DEFAULT 'Billed to Ledger',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 9. TRANSACTIONS TABLE
CREATE TABLE transactions (
    id SERIAL PRIMARY KEY,
    txn_id VARCHAR(50) UNIQUE NOT NULL,
    ref_code VARCHAR(100),
    payer VARCHAR(255),
    property_location TEXT,
    category VARCHAR(100),
    direction VARCHAR(20) CHECK (direction IN ('INFLOW', 'OUTFLOW')),
    amount NUMERIC(12, 2) NOT NULL,
    gross_amount VARCHAR(50),
    channel VARCHAR(100),
    status VARCHAR(50) DEFAULT 'Verified',
    receipt_attachment TEXT,
    timestamp VARCHAR(100),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 10. DOCUMENTS TABLE (Legal Vault)
CREATE TABLE documents (
    id SERIAL PRIMARY KEY,
    doc_id VARCHAR(50) UNIQUE NOT NULL,
    title VARCHAR(255) NOT NULL,
    file_type VARCHAR(100),
    entity_name VARCHAR(255),
    entity_sub TEXT,
    type VARCHAR(100),
    uploader VARCHAR(255),
    date VARCHAR(100),
    status VARCHAR(50) DEFAULT 'Active',
    notes TEXT,
    file_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 11. INSPECTIONS TABLE
CREATE TABLE inspections (
    id SERIAL PRIMARY KEY,
    inspection_id VARCHAR(50) UNIQUE NOT NULL,
    unit_name VARCHAR(100),
    property_info TEXT,
    tenant VARCHAR(255),
    type VARCHAR(50),
    date DATE,
    inspector VARCHAR(255),
    status VARCHAR(50) DEFAULT 'Completed',
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 12. MAINTENANCE TICKETS TABLE
CREATE TABLE maintenance_tickets (
    id SERIAL PRIMARY KEY,
    ticket_id VARCHAR(50) UNIQUE NOT NULL,
    title VARCHAR(255) NOT NULL,
    category VARCHAR(100),
    description TEXT,
    tenant_name VARCHAR(255),
    property_location TEXT,
    priority VARCHAR(50) DEFAULT 'Normal',
    technician VARCHAR(255),
    scheduled_time VARCHAR(100),
    status VARCHAR(50) DEFAULT 'Open',
    cost NUMERIC(12, 2) DEFAULT 0.00,
    date_filed VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 13. HANDOVERS TABLE
CREATE TABLE handovers (
    id SERIAL PRIMARY KEY,
    handover_id VARCHAR(50) UNIQUE NOT NULL,
    lease_id VARCHAR(50),
    tenant_name VARCHAR(255),
    property VARCHAR(255),
    unit TEXT,
    move_in_date DATE,
    score INT DEFAULT 100,
    main_keys INT DEFAULT 2,
    mail_keys INT DEFAULT 1,
    rfid_count INT DEFAULT 2,
    remotes_count INT DEFAULT 1,
    keys_issued VARCHAR(255),
    elec_baseline NUMERIC(12, 2),
    water_baseline NUMERIC(12, 2),
    status VARCHAR(50) DEFAULT 'Completed',
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 14. SETTLEMENTS TABLE
CREATE TABLE settlements (
    id SERIAL PRIMARY KEY,
    settlement_id VARCHAR(50) UNIQUE NOT NULL,
    lease_id VARCHAR(50),
    tenant_name VARCHAR(255),
    property_name VARCHAR(255),
    unit_no TEXT,
    vacate_date DATE,
    initial_deposit NUMERIC(12, 2),
    deductions NUMERIC(12, 2),
    net_refund NUMERIC(12, 2),
    status VARCHAR(50) DEFAULT 'Pending Audit',
    payment_method VARCHAR(100),
    reference_code VARCHAR(100),
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 15. NOTIFICATIONS TABLE
CREATE TABLE notifications (
    id VARCHAR(50) PRIMARY KEY,
    pov VARCHAR(50) DEFAULT 'all',
    category VARCHAR(50),
    status VARCHAR(20) DEFAULT 'unread',
    is_read BOOLEAN DEFAULT FALSE,
    title VARCHAR(255) NOT NULL,
    time VARCHAR(100),
    description TEXT,
    property TEXT,
    tag VARCHAR(100),
    tag_color VARCHAR(50),
    amount VARCHAR(50),
    action_url VARCHAR(255),
    action_text VARCHAR(100),
    urgent BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
